## 1. Authority and precedence

## 2. Sol review severity / blocking policy

The reviewer must distinguish CONTRACT-BLOCKING findings from NON-BLOCKING quality findings.

A finding may BLOCK the current step only if at least one is true:

1. It violates an explicit requirement of the authoritative Astra plan.
2. It makes a machine-verifiable factual claim false.
3. It weakens a safety, isolation, scope, rollback, resume, budget, test, review, or commit invariant.
4. It leaves required evidence missing, ambiguous, stale, or unbound to the reviewed candidate.
5. It creates behavior that can affect execution or release correctness.

The following are NON-BLOCKING unless one of those conditions applies: wording/style, formatting, redundant prose, naming preferences not required by contract, documentation polish, optional refactoring, speculative future hardening, and future-step requirements that are not current-step prerequisites.

If all explicit current-step requirements pass and only non-blocking findings remain, return `verdict=APPROVE`, `step_status=PASS`, and `authorized_for_next_step=true`; list remaining items under `non_blocking_findings`. Do not reject a completed step for future-step work.

Every blocking finding MUST include: `finding_id`, `severity`, `blocking_class`, `contract_source`, `exact_requirement`, `concrete_evidence`, `observed_state`, `expected_state`, `why_it_blocks_current_step`, `required_correction`, and `correction_class`.

Allowed blocking classes: `EXPLICIT_CONTRACT_VIOLATION`, `FACTUAL_INACCURACY`, `SAFETY_INVARIANT_VIOLATION`, `REQUIRED_EVIDENCE_GAP`, `EXECUTION_OR_RELEASE_CORRECTNESS`.

## 3. Non-blocking findings

Use a separate `non_blocking_findings` collection. Each item may include `finding_id`, `status=NON_BLOCKING`, `severity`, `issue`, `rationale`, `suggested_followup`, and `suggested_future_step`. Non-blocking findings MUST NOT change APPROVE to REJECT or silently become prerequisites without higher-authority contract text.

## 4. Micro-fix policy

A bounded defect confined to authorized files, mechanically correctable, and not altering architecture, security boundaries, runtime behavior, or scope should be classified `MICRO_FIX`. Examples include factual documentation errors, stale symbol/path names, missing required documentation entries whose source is known, and small evidence metadata corrections.

Sol MUST NOT edit the candidate. It must return the exact file, incorrect or missing claim, exact correction, and verification command/evidence. Hermes may apply the bounded correction.

After a micro-fix Hermes may request `review_mode=DELTA_ONLY_RECHECK`. The recheck verifies targeted findings resolved, no regression of prior PASS requirements, authorized path boundary, fresh candidate identity/hashes, and affected deterministic verification. Unrelated architecture review is not repeated.

def string_list(value: Any, field: str, *, allow_empty: bool = False) -> list[str]:
    if not isinstance(value, list) or (not value and not allow_empty):
        raise FactoryError(f"{field} must be a non-empty JSON array", "contract")

    def _stage(self, worktree: Path, allowed: list[str], run_dir: Path, sequence: int) -> tuple[str, str, list[str]]:
        git(worktree, "reset", "--quiet")
        changed = self._scope_check(worktree, allowed, f"prestage_{sequence}", run_dir)
        if not changed:
            raise FactoryError("builder produced no changes", "builder")
        git(worktree, "add", "--", *allowed)
        diff_text = git(worktree, "diff", "--cached", "--binary", "--no-ext-diff").stdout
        if not diff_text:
            raise FactoryError("staged candidate diff is empty", "git")
        digest = hashlib.sha256(diff_text.encode("utf-8")).hexdigest()
        write_text(run_dir / f"candidate_{sequence}.diff", diff_text)
        atomic_json(run_dir / f"candidate_{sequence}.json", {"sha256": digest, "changed_files": changed})
        return diff_text, digest, changed

    def _candidate_source_context(
        self,
        worktree: Path,
        allowed: list[str],
        changed: list[str],
        run_dir: Path,
        sequence: int,
    ) -> dict[str, Any]:
        """Build a bounded, read-only source snapshot for Sol's review.

        exact candidate that will be hash-checked, including newly created or
        deleted files. Planner and Luna transcripts are deliberately omitted.

        remaining = 80_000
        files: dict[str, Any] = {}
        for relative in sorted(set(allowed) | set(changed)):

            files[relative] = {
                "state": "present",
                "content": excerpt,
                "truncated": len(excerpt) < len(raw),
            }
            remaining -= len(excerpt)
            if remaining <= 0:
                break
        context = {
            "source": "staged index exact candidate state",
            "base_commit": git(worktree, "rev-parse", "HEAD").stdout.strip(),
            "files": files,
            "note": "Read-only context; omitted files were not included or exceeded the context limit.",
        }
        atomic_json(run_dir / f"candidate_source_context_{sequence}.json", context)
        return context

                raise FactoryError(f"git update-ref failed: {ref_res.stderr}", "checkpoint")

        # 2. Capture diffs with returncode validation
        unstaged_res = command(["git", "diff"], cwd=worktree)
        if unstaged_res.returncode != 0:
            raise FactoryError(f"git diff failed: {unstaged_res.stderr}", "checkpoint")
        staged_res = command(["git", "diff", "--cached"], cwd=worktree)
        if staged_res.returncode != 0:
            raise FactoryError(f"git diff --cached failed: {staged_res.stderr}", "checkpoint")

        # 3. Capture untracked files safely with NUL delimiter and base64 encoding
        untracked_res = command(["git", "ls-files", "-z", "--others", "--exclude-standard"], cwd=worktree)
        if untracked_res.returncode != 0:
            raise FactoryError(f"git ls-files failed: {untracked_res.stderr}", "checkpoint")

        raw_paths = untracked_res.stdout.split("\0")
        untracked_files: dict[str, dict[str, Any]] = {}
        for rel_p in raw_paths:
            if not rel_p:
                continue
            full_p = worktree / rel_p
            if full_p.is_symlink():
                untracked_files[rel_p] = {
                    "type": "symlink",
                    "target": os.readlink(full_p),
                }
            elif full_p.is_file():
                content_bytes = full_p.read_bytes()
                mode = full_p.stat().st_mode
                untracked_files[rel_p] = {
                    "type": "file",
                    "data": base64.b64encode(content_bytes).decode("ascii"),
                    "mode": oct(mode),
                }

        checkpoint = {
            "label": label,
            "created_at": now_iso(),
            "stash_commit": stash_commit,
            "stash_ref": stash_ref,
            "unstaged_diff": unstaged_res.stdout,
            "staged_diff": staged_res.stdout,
            "untracked_files": untracked_files,
        }
        atomic_json(run_dir / f"checkpoint_{label}.json", checkpoint)
        return checkpoint

    def status(self, identifier: str) -> dict[str, Any]:
        _, state, _ = self._load(identifier)
        return state

    def resume(self, identifier: str) -> dict[str, Any]:
        run_dir, state, config = self._load(identifier)
        if state["phase"] == "committed":
            return state
        if state["phase"] not in {"planned", "blocked", "failed", "worktree_ready"}:
            raise FactoryError(f"run cannot be resumed safely from phase {state['phase']!r}", "state")
        if not (run_dir / "sol_plan.json").is_file():
            repo = self._repo(config.repo)
            try:
                self._perform_planning(run_dir, state, config, repo)
            except FactoryError as exc:
                phase = "blocked" if exc.category in {"preflight", "provider"} else "failed"
                self._save(run_dir, state, phase, error=str(exc), error_category=exc.category)
                raise
        else:
            state["warm_start"] = True
            atomic_json(run_dir / "warm_start.json", {"warm_start": True, "resumed_from": state.get("phase"), "timestamp": now_iso()})
            atomic_json(run_dir / "run_state.json", state)
        return self.execute(identifier)


def split_files(raw_values: list[str] | None) -> list[str] | None:
    if not raw_values:
        return None
    values: list[str] = []
    for raw in raw_values:
        values.extend(item.strip() for item in raw.split(",") if item.strip())
    return normalized_paths(values)
